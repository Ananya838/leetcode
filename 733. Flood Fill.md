    def floodFill(self, image: list[list[int]], sr: int, sc: int, color: int) -> list[list[int]]:

        rows = len(image)
        cols = len(image[0])
        visited = [[False]*cols for _ in range(rows)]
        directions = [
            (0,-1),
            (0,1),
            (-1,0),
            (1,0)
        ]
        def  bfs(row, col, ans):

            visited[row][col] = True
            queue = deque([(row,col)])
            

            while queue:

                pr , pc = queue.popleft()

                for dr, dc in directions:

                    nr = pr + dr
                    nc = pc + dc

                    if (nr < 0 or nr >= rows) or (nc < 0 or nc >= cols):
                        continue
                    
                    if image[nr][nc] == ans and not visited[nr][nc]:
                        image[nr][nc] = color
                        queue.append((nr,nc))
                        visited[nr][nc] = True

from collections import deque

class Solution:
    def floodFill(self, image, sr, sc, color):
        rows = len(image)
        cols = len(image[0])

        old_color = image[sr][sc]

        # Already the required color
        if old_color == color:
            return image

        directions = [
            (0, -1),   # left
            (0, 1),    # right
            (-1, 0),   # up
            (1, 0)     # down
        ]

        queue = deque([(sr, sc)])
        image[sr][sc] = color

        while queue:
            r, c = queue.popleft()

            for dr, dc in directions:
                nr = r + dr
                nc = c + dc

                # Check boundary
                if nr < 0 or nr >= rows or nc < 0 or nc >= cols:
                    continue

                # Only fill cells having the original color
                if image[nr][nc] == old_color:
                    image[nr][nc] = color
                    queue.append((nr, nc))

        return image
