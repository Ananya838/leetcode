class Solution:
    def canVisitAllRooms(self, rooms: list[list[int]]) -> bool:

        # We use a set to remember which rooms we have already visited
        visited = set()

        # We start from room 0 because room 0 is unlocked initially
        stack = [0]

        # Continue until there are no more rooms to explore
        while stack:

            # Take one room from the stack
            node = stack.pop()

            # If we have already visited this room,
            # don't process it again
            if node in visited:
                continue

            # Mark the current room as visited
            visited.add(node)

            # rooms[node] gives us all the keys inside this room
            # Each key represents another room that we can enter
            for neighbour in rooms[node]:

                # If we haven't visited that room yet,
                # add it to the stack so we can visit it later
                if neighbour not in visited:
                    stack.append(neighbour)

        # If the number of visited rooms equals the total number
        # of rooms, then we were able to visit every room
        return len(visited) == len(rooms)
