class Solution:
    def minAddToMakeValid(self, s: str) -> int:
        stack = []
        ans = 0
        for ch in s:

            if ch == '(':
                stack.append(ch)
                ans += 1
            elif ch == ')':
                if stack and stack[-1] == '(':
                    stack.pop()
                    ans-=1
                else:
                    stack.append(ch)
                    ans+=1
        return ans

        for ch in s:
